  <script>
    var fileCount = 0;

    function addFileUploader() {
      if (fileCount < 5) {
        var fileInput = document.createElement("input");
        fileInput.type = "file";
        fileInput.className = "form-control";

        var lineBreak1 = document.createElement("br");

        if (fileCount > 0) {
          document.getElementById("fileContainer").appendChild(lineBreak1);
        }

        document.getElementById("fileContainer").appendChild(fileInput);

        fileCount++;

        // Enable the button if the file count is less than 5
        if (fileCount < 5) {
          document.getElementById("addButton").disabled = false;
        } else {
          // Disable the button if the file count is 5
          document.getElementById("addButton").disabled = true;
        }
      }
    }
  </script>
